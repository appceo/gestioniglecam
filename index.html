async function buscarRegistros() { const criterio = document.getElementById('criterioBusqueda').value; const valor = document.getElementById('valorBusqueda').value.trim(); let query = supabase.from('miembros').select('*'); if (valor !== '') { if (criterio === 'identificacion') { // Convierte a número si es posible const num = parseInt(valor); if (!isNaN(num)) { query = query.eq('identificacion', num); } else { query = query.ilike('identificacion', `%${valor}%`); } } else if (criterio === 'telefono') { query = query.ilike('telefono_movil', `%${valor}%`); } else if (criterio === 'iglesia') { query = query.ilike('iglesia', `%${valor}%`); } else if (criterio === 'ciudad_iglesia') { query = query.ilike('ciudad_iglesia', `%${valor}%`); } else if (criterio === 'direccion_iglesia') { query = query.ilike('direccion_iglesia', `%${valor}%`); } else if (criterio === 'nombres_apellidos') { // Separa el texto introducido por espacios para permitir búsquedas compuestas (ej: "Javier Aponte") const palabras = valor.split(/\s+/).filter(p => p.length > 0); // Construye la cláusula OR compatible con Supabase PostgREST palabras.forEach(p => { const orCondicion = `primer_nombre.ilike.%${p}%,segundo_nombre.ilike.%${p}%,primer_apellido.ilike.%${p}%,segundo_apellido.ilike.%${p}%`; query = query.or(orCondicion); }); } } const { data, error } = await query; if (error) { console.error("Error en Supabase:", error); alert('Error al consultar datos: ' + error.message); return; } resultadosActuales = data; renderizarTabla(data); }
